module.exports = async function handler(req, res) { res.setHeader('Access-Control-Allow-Origin', '*'); res.setHeader('Access-Control-Allow-Methods', 'GET'); const { startX, startY, endX, endY } = req.query; if (!startX || !startY || !endX || !endY) return res.status(400).json({ error: '좌표 필요' }); const key = process.env.ODSAY_KEY; if (!key) return res.status(500).json({ error: 'ODSAY_KEY 없음' }); try { // + 기호를 %2B로 인코딩 (URL에서 안전하게) const encodedKey = key.replace(/\+/g, '%2B'); const url = `https://api.odsay.com/v1/api/searchPubTransPathT` + `?SX=${startX}&SY=${startY}` + `&EX=${endX}&EY=${endY}` + `&apiKey=${encodedKey}`; const r = await fetch(url); const text = await r.text(); return res.status(200).json({ ok: true, debug: true, status: r.status, raw: text.slice(0, 3000) }); } catch (e) { return res.status(500).json({ ok: false, error: e.message }); } };
